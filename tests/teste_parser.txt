// Programa valido para o parser: ./parser_exe < tests/teste_parser.txt
int soma = 0;
float media;
char letra = 'a';
long grande = 100;

for (int i = 0; i < 10; i = i + 1) {
    soma = soma + i;
}

while (soma > 0) {
    soma = soma - 1;
    if (soma == 5) break;
}

do {
    soma = soma + 1;
} while (soma < 3);

if (soma >= 3 && !(letra == 'b'))
    media = soma / 2;
else
    media = -1;

switch (soma) {
    case 1:
        soma = 10;
        break;
    case 2:
        soma = 20;
        break;
}

return soma;
